        [MenuItem("FruitGame/Build Scene")]
        public static void Build()
        {
            // Open the new scene first: NewScene unloads unused assets, which would orphan
            // ScriptableObjects created just before it.
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);

            EnsureFolder(MaterialDir);
            EnsureFolder(FruitTypeDir);

            var config = LoadOrCreate<GameConfig>(ConfigPath);
            var types = CreateFruitTypes();
            var lit = LitMaterial("Greybox_Lit", Color.white);
            var font = CreateFontAsset();
            var sticker = CreateStickerMaterial(font);

            // Cameras: a black background camera clears the letterbox bars, the main camera renders 3:4.
            var bgCamGo = new GameObject("Letterbox Camera");
            var bgCam = bgCamGo.AddComponent<Camera>();
            bgCam.clearFlags = CameraClearFlags.SolidColor;
            bgCam.backgroundColor = Color.black;
            bgCam.cullingMask = 0;
            bgCam.depth = -10;

            var camGo = new GameObject("Main Camera") { tag = "MainCamera" };
            var cam = camGo.AddComponent<Camera>();
            cam.transform.position = new Vector3(0f, 0f, -12f);
            cam.fieldOfView = 30f;
            cam.nearClipPlane = 0.3f;
            cam.farClipPlane = 50f;
            cam.clearFlags = CameraClearFlags.SolidColor;
            cam.backgroundColor = Sky;
            cam.depth = 0;
            camGo.AddComponent<AudioListener>();
            camGo.AddComponent<PortraitLetterbox>();
            camGo.AddComponent<UniversalAdditionalCameraData>().renderPostProcessing = true;
            BuildPostProcessing();

            var lightGo = new GameObject("Directional Light");
            var light = lightGo.AddComponent<Light>();
            light.type = LightType.Directional;
            light.color = Hex("#FFF1DC");
            light.intensity = 1.2f;
            light.shadows = LightShadows.Soft;
            lightGo.transform.rotation = Quaternion.Euler(50f, 30f, 0f);
            RenderSettings.ambientMode = AmbientMode.Trilight;
            RenderSettings.ambientSkyColor = Hex("#CFE3EE");
            RenderSettings.ambientEquatorColor = Hex("#B9D3B0");
            RenderSettings.ambientGroundColor = Hex("#7E8B6A");

            // Layers, back to front (README §2). L3–L5 are placeholders until M4/M5.
            var l5 = new GameObject("L5_Bokeh").transform;
            var bokeh = Primitive(PrimitiveType.Quad, "Bokeh (placeholder)", l5, LitMaterial("Greybox_Bokeh", Hex("#B9D3DD")));
            bokeh.position = new Vector3(0f, 0f, 6f);
            var bokehRenderer = bokeh.GetComponent<Renderer>();
            bokehRenderer.shadowCastingMode = ShadowCastingMode.Off;
            bokehRenderer.receiveShadows = false;
            bokeh.localScale = new Vector3(4.8f * 18f / 12f + 0.2f, 6.4f * 18f / 12f + 0.2f, 1f);

            var far = new GameObject("L4b_FarTrees").transform;
            BuildBackgroundTrees(far);

            var l4 = new GameObject("L4_Pile").transform;
            var floor = Primitive(PrimitiveType.Cube, "Pile Ground", l4, LitMaterial("Greybox_Ground", Hex("#B7A58A")));
            floor.position = new Vector3(0f, config.pileFloorY - 0.3f, config.pileZ);
            floor.localScale = new Vector3(8f, 0.6f, 1.6f);

            var l3 = new GameObject("L3_Person").transform;
            var personMat = LitMaterial("Greybox_Person", Hex("#5A6B7D"));
            var torso = Primitive(PrimitiveType.Capsule, "Torso (placeholder)", l3, personMat);
            torso.position = new Vector3(0f, -3.4f, 2f);
            torso.localScale = new Vector3(2.4f, 1.6f, 0.6f);
            var head = Primitive(PrimitiveType.Sphere, "Head (placeholder)", l3, LitMaterial("Greybox_Skin", Hex("#EFCAA8")));
            head.position = new Vector3(0f, -0.5f, 2f);
            head.localScale = new Vector3(1.1f, 1.35f, 0.8f);

            var l2 = new GameObject("L2_Play").transform;
            var basket = BuildBasket(l2, config);
            var assist = basket.gameObject.AddComponent<BasketAssist>();
            Set(assist, "config", config);
            Set(assist, "basket", basket);

            var spawnerGo = new GameObject("FruitSpawner");
            spawnerGo.transform.SetParent(l2, false);
            var spawner = spawnerGo.AddComponent<FruitSpawner>();
            Set(spawner, "config", config);
            Set(spawner, "basket", basket);
            Set(spawner, "assist", assist);
            Set(spawner, "greyboxMaterial", lit);
            SetList(spawner, "fruitTypes", types);

            var l1 = new GameObject("L1_Foreground").transform;
            BuildForeground(l1);

            // Systems
            var systems = new GameObject("Systems");
            var input = systems.AddComponent<FallbackInput>();
            Set(input, "config", config);
            Set(input, "worldCamera", cam);
            Set(basket, "input", input);

            var audioGo = new GameObject("Audio");
            audioGo.transform.SetParent(systems.transform, false);
            var src = audioGo.AddComponent<AudioSource>();
            src.playOnAwake = false;
            src.spatialBlend = 0f;
            var sfx = audioGo.AddComponent<AudioFeedback>();
            Set(sfx, "catchPop", Clip("catch_pop"));
            Set(sfx, "rimTink", Clip("rim_tink"));
            Set(sfx, "missThud", Clip("miss_thud"));
            Set(sfx, "spawnRustle", Clip("spawn_rustle"));
            Set(sfx, "pileLand", Clip("pile_land"));
            Set(sfx, "countdownTick", Clip("countdown_tick"));
            Set(sfx, "roundOver", Clip("round_over"));

            var pileGo = new GameObject("Fruit Pile");
            pileGo.transform.SetParent(l4, false);
            pileGo.transform.position = new Vector3(0f, 0f, config.pileZ);
            var pile = pileGo.AddComponent<FruitPile>();
            Set(pile, "config", config);
            Set(pile, "spawner", spawner);
            Set(pile, "audioFeedback", sfx);
            EnsureLayer("Pile");

            var gsm = systems.AddComponent<GameStateMachine>();
            var hud = BuildHud(cam, font, sticker, gsm);

            Set(gsm, "config", config);
            Set(gsm, "spawner", spawner);
            Set(gsm, "basket", basket);
            Set(gsm, "fallbackInput", input);
            Set(gsm, "hud", hud);
            Set(gsm, "audioFeedback", sfx);
            Set(gsm, "pile", pile);

            EditorSceneManager.SaveScene(scene, ScenePath);
            AddToBuildSettings(ScenePath);
            ConfigurePlayer();
            AssetDatabase.SaveAssets();
            Debug.Log($"[FruitGame] Scene built at {ScenePath}");
        }

        static BasketController BuildBasket(Transform parent, GameConfig config)
        {
            var root = new GameObject("Basket").transform;
            root.SetParent(parent, false);
            root.position = new Vector3(0f, config.rimY, 0f);

            var visual = new GameObject("Visual").transform;
            visual.SetParent(root, false);

            // Low-poly basket model (3D Assets/FruitGame Extras/Basket). It sits in a "Model" child tilted
            // toward the camera so the mouth reads; "Visual" carries the squash/wobble animation.
            var model = new GameObject("Model").transform;
            model.SetParent(visual, false);
            model.localRotation = Quaternion.Euler(-5f, 0f, 0f);
            // The model is authored with a 1.1 m mouth and 0.65 m height; scale it to the configured size
            // so it matches the colliders BasketController builds from the same config.
            float widthScale = config.basketMouthWidth / 1.1f;
            model.localScale = new Vector3(widthScale, config.basketHeight / 0.65f, widthScale);
            var basketPrefab = ArtLibrary.Basket();
            if (basketPrefab != null)
                PrefabUtility.InstantiatePrefab(basketPrefab, model);

            var rb = root.gameObject.AddComponent<Rigidbody>();
            rb.isKinematic = true;
            rb.useGravity = false;
            var basket = root.gameObject.AddComponent<BasketController>();
            Set(basket, "config", config);
            Set(basket, "visual", visual);
            return basket;
        }

        /// <summary>A gentle global grade: a touch of saturation and contrast, soft bloom and vignette (no tonemapping; it greys the pastels).</summary>
        static void BuildPostProcessing()
        {
            EnsureFolder(Root + "/Art/Rendering");
            string path = Root + "/Art/Rendering/GamePostProfile.asset";
            var profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(path);
            if (profile == null)
            {
                profile = ScriptableObject.CreateInstance<VolumeProfile>();
                AssetDatabase.CreateAsset(profile, path);
            }

            T Override<T>() where T : VolumeComponent
            {
                if (!profile.TryGet<T>(out var c))
                {
                    c = profile.Add<T>(true);
                    c.name = typeof(T).Name;
                    AssetDatabase.AddObjectToAsset(c, profile);
                }
                c.active = true;
                return c;
            }

            var tone = Override<Tonemapping>();
            tone.mode.Override(TonemappingMode.None);
            var color = Override<ColorAdjustments>();
            color.saturation.Override(10f);
            color.contrast.Override(6f);
            var bloom = Override<Bloom>();
            bloom.intensity.Override(0.25f);
            bloom.threshold.Override(1.0f);
            var vignette = Override<Vignette>();
            vignette.intensity.Override(0.2f);
            vignette.smoothness.Override(0.45f);
            EditorUtility.SetDirty(profile);

            var go = new GameObject("Post Processing");
            var volume = go.AddComponent<Volume>();
            volume.isGlobal = true;
            volume.sharedProfile = profile;
        }

        /// <summary>
        /// Distant canopies behind the person and the pile, hazed toward light blue so they read as far away.
        /// Their crowns still frame the top corners.
        /// </summary>
        static void BuildBackgroundTrees(Transform parent)
        {
            var haze = Hex("#A9C6DD");
            var oak = ArtLibrary.FarTree("003_Oak_Tree", haze, 0.7f);
            var pine = ArtLibrary.FarTree("002_Pine_Tree", haze, 0.7f);
            var placed = new[]
            {
                ArtLibrary.Place(oak, parent, new Vector3(-5.0f, -0.9f, 5.2f), 7.2f, 20f),
                ArtLibrary.Place(oak, parent, new Vector3(5.0f, -1.1f, 5.2f), 7.5f, -70f, true),
                ArtLibrary.Place(pine, parent, new Vector3(-3.4f, -4.6f, 5.4f), 4.2f, 30f),
                ArtLibrary.Place(pine, parent, new Vector3(3.5f, -4.6f, 5.4f), 4.6f, 200f),
            };
            // Distant trees neither cast shadows (they'd darken the webcam backdrop) nor receive them.
            foreach (var t in placed)
                foreach (var r in t.GetComponentsInChildren<Renderer>())
                {
                    r.shadowCastingMode = ShadowCastingMode.Off;
                    r.receiveShadows = false;
                }
        }

        static RectTransform MakeRect(string name, Transform parent, Vector2 anchorMin, Vector2 anchorMax)
        {
            var go = new GameObject(name, typeof(RectTransform));
            go.transform.SetParent(parent, false);
            var rt = (RectTransform)go.transform;
            rt.anchorMin = anchorMin;
            rt.anchorMax = anchorMax;
            rt.offsetMin = rt.offsetMax = Vector2.zero;
            return rt;
        }

        /// <summary>A large, mouse-friendly button whose click calls a GameStateMachine method (saved in the scene).</summary>
        static Button MakeButton(string name, Transform parent, TMP_FontAsset font, string label, float fontSize,
            Vector2 anchor, Vector2 pos, Vector2 size, Color color, UnityAction onClick)
        {
            var go = new GameObject(name, typeof(RectTransform), typeof(Image), typeof(Button));
            go.transform.SetParent(parent, false);
            var rt = (RectTransform)go.transform;
            rt.anchorMin = rt.anchorMax = anchor;
            rt.pivot = new Vector2(0.5f, anchor.y >= 1f ? 1f : 0.5f);
            rt.anchoredPosition = pos;
            rt.sizeDelta = size;

            var img = go.GetComponent<Image>();
            img.sprite = RoundedSprite();
            img.type = Image.Type.Sliced;
            img.color = color;

            var button = go.GetComponent<Button>();
            var colors = button.colors;
            colors.normalColor = Color.white;
            colors.highlightedColor = new Color(1.12f, 1.12f, 1.12f, 1f);
            colors.pressedColor = new Color(0.8f, 0.8f, 0.8f, 1f);
            colors.fadeDuration = 0.08f;
            button.colors = colors;
            UnityEventTools.AddPersistentListener(button.onClick, onClick);

            var text = Text("Label", go.transform, font, null, fontSize, TextAlignmentOptions.Center,
                new Vector2(0.5f, 0.5f), Vector2.zero, size, label);
            text.color = Color.white;
            return button;
        }

        static Sprite RoundedSprite() => AssetDatabase.GetBuiltinExtraResource<Sprite>("UI/Skin/UISprite.psd");

        static TextMeshProUGUI Text(string name, Transform parent, TMP_FontAsset font, Material mat, float size,
            TextAlignmentOptions align, Vector2 anchor, Vector2 pos, Vector2 box, string text)
        {
            var go = new GameObject(name, typeof(RectTransform));
            go.transform.SetParent(parent, false);
            var rt = (RectTransform)go.transform;
            rt.anchorMin = rt.anchorMax = anchor;
            rt.pivot = anchor;
            rt.anchoredPosition = pos;
            rt.sizeDelta = box;
            var t = go.AddComponent<TextMeshProUGUI>();
            if (font != null) t.font = font;
            if (mat != null) t.fontSharedMaterial = mat;
            t.fontSize = size;
            t.alignment = align;
            t.color = Color.white;
            t.text = text;
            t.raycastTarget = false;
            return t;
        }

        static TMP_FontAsset CreateFontAsset()
        {
            var existing = AssetDatabase.LoadAssetAtPath<TMP_FontAsset>(FontAssetPath);
            if (existing != null) return existing;
            var ttf = AssetDatabase.LoadAssetAtPath<Font>(FontTtf);
            if (ttf == null) { Debug.LogWarning("[FruitGame] Fredoka-Bold.ttf not found; using TMP default font."); return null; }

            var fa = TMP_FontAsset.CreateFontAsset(ttf, 90, 9, GlyphRenderMode.SDFAA, 1024, 1024, AtlasPopulationMode.Dynamic);
            fa.name = "Fredoka-Bold SDF";
            AssetDatabase.CreateAsset(fa, FontAssetPath);
            fa.atlasTextures[0].name = "Fredoka-Bold Atlas";
            AssetDatabase.AddObjectToAsset(fa.atlasTextures[0], fa);
            fa.material.name = "Fredoka-Bold Material";
            AssetDatabase.AddObjectToAsset(fa.material, fa);
            fa.TryAddCharacters("0123456789:+×!?.,'- ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz←→");
            EditorUtility.SetDirty(fa);
            AssetDatabase.SaveAssets();
            return fa;
        }

        static Material CreateStickerMaterial(TMP_FontAsset font)
        {
            if (font == null) return null;
            var mat = AssetDatabase.LoadAssetAtPath<Material>(StickerMaterialPath);
            if (mat == null)
            {
                mat = new Material(font.material) { name = "Fredoka-Bold Sticker" };
                AssetDatabase.CreateAsset(mat, StickerMaterialPath);
            }
            // White fill, thick magenta outline, hard drop shadow down-right: the "sticker" look (README §9).
            // The desktop SDF shader is used because the mobile one drops the outline.
            var sdf = Shader.Find("TextMeshPro/Distance Field");
            if (sdf != null) mat.shader = sdf;
            mat.EnableKeyword("OUTLINE_ON");
            mat.SetFloat("_FaceDilate", 0.1f);
            mat.SetColor("_OutlineColor", ScorePink);
            mat.SetFloat("_OutlineWidth", 0.32f);
            mat.EnableKeyword("UNDERLAY_ON");
            mat.SetColor("_UnderlayColor", new Color(0.35f, 0.04f, 0.18f, 0.9f));
            mat.SetFloat("_UnderlayOffsetX", 0.6f);
            mat.SetFloat("_UnderlayOffsetY", -0.6f);
            mat.SetFloat("_UnderlayDilate", 0.4f);
            mat.SetFloat("_UnderlaySoftness", 0f);
            EditorUtility.SetDirty(mat);
            return mat;
        }

        static List<FruitType> CreateFruitTypes()
        {
            var list = new List<FruitType>();
            foreach (var f in Fruits)
            {
                var t = LoadOrCreate<FruitType>($"{FruitTypeDir}/{f.name}.asset");
                t.id = f.id;
                t.displayName = f.name;
                t.points = 1;
                t.spawnWeight = f.weight;
                t.sizeScale = f.size;
                t.colliderShape = f.shape;
                t.greyboxColor = Hex(f.color);
                t.visualPrefabs = ArtLibrary.FruitPrefabs(f.name);
                EditorUtility.SetDirty(t);
                list.Add(t);
            }
            return list;
        }

        static Material LitMaterial(string name, Color color)
        {
            string path = $"{MaterialDir}/{name}.mat";
            var mat = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (mat == null)
            {
                var shader = Shader.Find("Universal Render Pipeline/Lit");
                mat = new Material(shader) { name = name };
                AssetDatabase.CreateAsset(mat, path);
            }
            mat.SetColor("_BaseColor", color);
            mat.SetFloat("_Smoothness", 0.15f);
            EditorUtility.SetDirty(mat);
            return mat;
        }

        static Transform Primitive(PrimitiveType type, string name, Transform parent, Material mat)
        {
            var go = GameObject.CreatePrimitive(type);

            go.GetComponent<Renderer>().sharedMaterial = mat;

        }

        static void EnsureLayer(string name)
        {
            if (LayerMask.NameToLayer(name) >= 0) return;
            var tagManager = new SerializedObject(AssetDatabase.LoadAllAssetsAtPath("ProjectSettings/TagManager.asset")[0]);
            var layers = tagManager.FindProperty("layers");
            for (int i = 8; i < layers.arraySize; i++)
            {
                var p = layers.GetArrayElementAtIndex(i);
                if (!string.IsNullOrEmpty(p.stringValue)) continue;
                p.stringValue = name;
                tagManager.ApplyModifiedPropertiesWithoutUndo();
                return;
            }
            Debug.LogWarning($"[FruitGame] No free layer for '{name}'.");
        }

        static void AddToBuildSettings(string path)
        {
            var scenes = new List<EditorBuildSettingsScene>(EditorBuildSettings.scenes);
            scenes.RemoveAll(s => s.path == path);
            scenes.Insert(0, new EditorBuildSettingsScene(path, true));
            EditorBuildSettings.scenes = scenes.ToArray();
        }

        static T LoadOrCreate<T>(string path) where T : ScriptableObject
        {
            var asset = AssetDatabase.LoadAssetAtPath<T>(path);
            if (asset != null) return asset;
            asset = ScriptableObject.CreateInstance<T>();
            AssetDatabase.CreateAsset(asset, path);
            return asset;
        }

        static void EnsureFolder(string path)
        {
            if (AssetDatabase.IsValidFolder(path)) return;
            int slash = path.LastIndexOf('/');
            EnsureFolder(path.Substring(0, slash));
            AssetDatabase.CreateFolder(path.Substring(0, slash), path.Substring(slash + 1));
        }

        static void Set(Object target, string field, Object value)
        {
            var so = new SerializedObject(target);
            var p = so.FindProperty(field);
            if (p == null) { Debug.LogError($"[FruitGame] {target.GetType().Name}.{field} not found"); return; }
            p.objectReferenceValue = value;
            so.ApplyModifiedPropertiesWithoutUndo();
        }

        static void SetList<T>(Object target, string field, List<T> values) where T : Object
        {
            var so = new SerializedObject(target);
            var p = so.FindProperty(field);
            p.arraySize = values.Count;
            for (int i = 0; i < values.Count; i++) p.GetArrayElementAtIndex(i).objectReferenceValue = values[i];
            so.ApplyModifiedPropertiesWithoutUndo();
        }

        static Color Hex(string hex) => ColorUtility.TryParseHtmlString(hex, out var c) ? c : Color.magenta;
    }
